\documentclass[tikz,border=10pt]{standalone}
\usepackage{amsmath}

\begin{document}

\begin{tikzpicture}

    % 定义函数参数
    \def\a{3}
    \def\b{8}
    \def\c{10}
    
    % 绘制函数
    \draw[domain=-3:3,smooth,variable=\x,blue,thick] plot ({\x},{\a*\x^2 + \b*\x + \c}) node[right] {$f(x) = 3x^2 + 8x + 10$};

    % 绘制导数（梯度）箭头
    \foreach \x in {-2, -1, 0, 1, 2} {
        \pgfmathsetmacro{\y}{\a*\x^2 + \b*\x + \c} % 计算函数值
        \pgfmathsetmacro{\grad}{2*\a*\x + \b} % 计算导数值
        
        % 绘制箭头
        \draw[->, red] (\x, \y) -- ++(1, \grad) node[above] {$\nabla f(x)$};
    }

    % 坐标轴
    \draw[->] (-4,0) -- (4,0) node[right] {$x$};
    \draw[->] (0,-10) -- (0,40) node[above] {$f(x)$};

\end{tikzpicture}

\end{document}
